cmake_minimum_required(VERSION 3.16)
project(m80_field_probe LANGUAGES CXX)
option(M80_STATE_PROBE_ONLY "Build only the read-only state probe with the installed compatible SDK" OFF)
if(M80_STATE_PROBE_ONLY)
  find_package(Franka CONFIG REQUIRED)
else()
  find_package(Franka 0.18 CONFIG REQUIRED)
endif()
add_executable(m80_read_state read_state.cpp)
target_compile_features(m80_read_state PRIVATE cxx_std_17)
target_link_libraries(m80_read_state PRIVATE Franka::Franka)
if(M80_STATE_PROBE_ONLY)
  return()
endif()
add_executable(m80_model_audit model_audit.cpp)
target_compile_features(m80_model_audit PRIVATE cxx_std_17)
target_link_libraries(m80_model_audit PRIVATE Franka::Franka)
add_library(m80_joint_stage STATIC libfranka_stage.cpp)
target_compile_features(m80_joint_stage PUBLIC cxx_std_17)
target_include_directories(m80_joint_stage PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(m80_joint_stage PUBLIC Franka::Franka)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
add_executable(m80_franka_hand franka_hand.cpp)
target_compile_features(m80_franka_hand PRIVATE cxx_std_17)
target_link_libraries(m80_franka_hand PRIVATE Franka::Franka Threads::Threads)
add_executable(m80_site_motion site_motion.cpp)
target_link_libraries(m80_site_motion PRIVATE m80_joint_stage Eigen3::Eigen Threads::Threads)
add_executable(test_servo_core test_servo_core.cpp)
target_compile_features(test_servo_core PRIVATE cxx_std_17)
enable_testing()
add_executable(test_site_watchdog test_site_watchdog.cpp)
target_compile_features(test_site_watchdog PRIVATE cxx_std_17)
add_test(NAME site_watchdog_latched_timeout COMMAND test_site_watchdog)
add_test(NAME native_hand_offline_guards COMMAND ${CMAKE_COMMAND}
  -DPROGRAM=$<TARGET_FILE:m80_franka_hand> -P ${CMAKE_CURRENT_SOURCE_DIR}/check_hand_cli.cmake)
add_test(NAME site_motion_offline_guards COMMAND ${CMAKE_COMMAND}
  -DPROGRAM=$<TARGET_FILE:m80_site_motion> -P ${CMAKE_CURRENT_SOURCE_DIR}/check_site_cli.cmake)
add_test(NAME servo_core COMMAND test_servo_core)
add_executable(test_stage_link test_stage_link.cpp)
target_link_libraries(test_stage_link PRIVATE m80_joint_stage)
add_test(NAME stage_sdk_link COMMAND test_stage_link)
add_executable(test_field_context test_field_context.cpp)
target_link_libraries(test_field_context PRIVATE m80_joint_stage)
add_test(NAME field_context_abort_guards COMMAND test_field_context)
find_package(Threads REQUIRED)
add_executable(test_telemetry test_telemetry.cpp)
target_compile_features(test_telemetry PRIVATE cxx_std_17)
target_link_libraries(test_telemetry PRIVATE Threads::Threads)
add_test(NAME telemetry_spsc COMMAND test_telemetry)
set_tests_properties(telemetry_spsc PROPERTIES TIMEOUT 15)
add_test(NAME readonly_model_input_guards COMMAND ${CMAKE_COMMAND}
  -DPROGRAM=$<TARGET_FILE:m80_model_audit> -P ${CMAKE_CURRENT_SOURCE_DIR}/check_model_cli.cmake)
